CREATE TABLE Pracownicy(
	identyfikator_pracownika int PRIMARY KEY AUTO_INCREMENT,
	imie_pracownika VARCHAR(30) NOT NULL,
    nazwisko_pracownika VARCHAR(40) NOT NULL,
    numer_pesel int(11) UNIQUE,
    pensja DECIMAL(10,2) DEFAULT 4000.00,
    data_zatrudnienia DATE NOT NULL,
    typ_etatu enum('pełny','1/2','1/4') DEFAULT 'pełny',
    numer_telefonu VARCHAR(15) NULL,
);
CREATE TABLE Stanowiska(
identyfikator_stanowiska int PRIMARY KEY AUTO_INCREMENT,
nazwa_stanowiska VARCHAR(40) NOT NULL,
opis_stanowiska text,
stawka_godzinowa decimal(10,2) DEFAULT 0,
data_waznosci_stanowiska date
);
CREATE TABLE Projekty(
identyfikator_projektu int PRIMARY KEY AUTO_INCREMENT,
nazwa_projektu VARCHAR(50) NOT NULL,
budzet_projektu decimal(10,2) DEFAULT 10000,
data_rozpoczecia_projektu date NULL,
priorytet_projektu ENUM('niski', 'sredni', 'wysoki') DEFAULT 'sredni',
opis_projektu text NULL
);
CREATE TABLE Dzialy(
identyfikator_dzialu int PRIMARY KEY AUTO_INCREMENT,
nazwa_dzialu VARCHAR(30) NOT NULL,
numer_telefonu_dzialu VARCHAR(15) NULL,
budzet_roczny_dzialu decimal(10,2) DEFAULT 0,
data_utworzenia_dzialu date DEFAULT CURRENT_DATE
);
CREATE TABLE Premie(
identyfikator_premii int PRIMARY KEY AUTO_INCREMENT.
kwota_premii decimal(10,2) NOT NULL,
rok_przyznania_premii YEAR NOT NULL,
typ_premii ENUM('uznaniowa', 'stała', 'okolicznosciowa') DEFAULT 'uznaniowa',
data_wyplaty_premii date NULL
);
